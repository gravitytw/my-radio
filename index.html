hls.loadSource(url);
hls.attachMedia(audio);
hls.on(Hls.Events.MANIFEST_PARSED, () => audio.play().catch(() => {}));
} else if (audio.canPlayType('application/vnd.apple.mpegurl')) {
audio.src = url;
audio.addEventListener('loadedmetadata', () => audio.play().catch(() => {}));
}
} else {
audio.src = url;
audio.load();
audio.play().catch(() => {});
}
}
function addNewStation() {
const nameInput = document.getElementById('new-name');
const urlInput = document.getElementById('new-url');
const isHlsCheck = document.getElementById('new-ishls');
if (!nameInput.value.trim() || !urlInput.value.trim()) return alert('請填寫完整資訊');
stations.push({ name: nameInput.value.trim(), url: urlInput.value.trim(), isHls: isHlsCheck.checked });
localStorage.setItem('roy_radio_stations', JSON.stringify(stations));
renderStations();
nameInput.value = ''; urlInput.value = ''; isHlsCheck.checked = false;
}
function deleteStation(index) {
if (confirm(`確定要刪除「${stations[index].name}」嗎？`)) {
stations.splice(index, 1);
localStorage.setItem('roy_radio_stations', JSON.stringify(stations));
renderStations();
}
}
function changeBgUrl() {

if (url && url.trim() !== '') {
document.body.style.backgroundImage = `linear-gradient(rgba(0, 0, 0, 0.65), rgba(0, 0, 0, 0.8)), url('${url}')`;
localStorage.setItem('roy_radio_bg', url);
}
}
function changeBgFile(event) {
const file = event.target.files;
if (file) {
const reader = new FileReader();
reader.onload = function(e) {
const base64Url = e.target.result;
document.body.style.backgroundImage = `linear-gradient(rgba(0, 0, 0, 0.65), rgba(0, 0, 0, 0.8)), url('${base64Url}')`;
try { localStorage.setItem('roy_radio_bg', base64Url); } catch (e) { alert('本地圖片檔案較大，建議使用小於 1MB 圖片或直接貼上網路圖片網址。'); }
};
reader.readAsDataURL(file);
}
}
